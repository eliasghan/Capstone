The surrogate model, one plot per function¶

For each function there is a single figure: one panel per input dimension.

  • Bold blue line — the surrogate's prediction as that one input is swept from 0 to 1.
  • Blue band — how much that prediction moves when the other inputs change.
  • Grey dots — every observation, plotted at its own value of that input.

Why the dots don't sit on the line¶

The line has one height for each value of x0. A real observation has a y that depends on x0 and on every other input at the same time. Two runs can share the same x0 and return completely different values because their other coordinates differ. So a dot sitting above or below the line is not the model being wrong — it is the other dimensions doing their part.

The band is exactly that effect made visible. It is drawn by re-running the sweep with the other inputs held at many different settings taken from the real data, and shading between them. A dot inside the band is fully accounted for: some setting of the other inputs explains it. A dot far outside the band is the only kind that would signal a genuine problem.

Where the band is narrow, that input acts on its own and the line can be trusted by itself. Where the band is wide, the input's effect depends heavily on the others, and no single-dimension picture can tell the whole story.

import warnings, numpy as np, pandas as pd
import matplotlib.pyplot as plt
warnings.filterwarnings("ignore")
from bbo_toolkit import fit_gp

RANDOM_STATE = 0
N_GRID       = 200    # smoothness of each curve
N_BACKGROUND = 40     # settings of the other inputs used to build the band
TRANSFORM    = {1: "log10abs", 5: "log"}   # F1 and F5 need a scale change

MEAN, MARK, INK, INK2, GRID = "#2a78d6", "#eb6834", "#0b0b0b", "#52514e", "#e6e5e2"
plt.rcParams.update({
    "figure.facecolor": "#fcfcfb", "axes.facecolor": "#fcfcfb",

})

def plot_surrogate(i):
    """One figure for function i: the surrogate along each input dimension."""
    X = np.load(f"function_{i}/initial_inputs.npy")
    y = np.load(f"function_{i}/initial_outputs.npy").ravel()
    try:
        df = pd.read_csv(f"function_{i}/new_trials.csv")
        X = np.vstack([X, df.iloc[:, :-1].to_numpy(float)])
        y = np.concatenate([y, df.iloc[:, -1].to_numpy(float)])
    except (FileNotFoundError, pd.errors.EmptyDataError):
        pass

    t = TRANSFORM.get(i)
    yy, label = (np.log(y), "log y") if t == "log" else \
                (np.log10(np.abs(y)), "log10 |y|") if t == "log10abs" else (y, "y")

    gp = fit_gp(X, yy, random_state=RANDOM_STATE)
    d = X.shape[1]
    g = np.linspace(0, 1, N_GRID)
    rng = np.random.default_rng(RANDOM_STATE)
    bg = X[rng.choice(len(X), size=min(N_BACKGROUND, len(X)), replace=False)]
    best = X[int(np.argmax(yy))]

    curves = []
    for j in range(d):
        S = np.empty((len(bg), N_GRID))
        for r, b in enumerate(bg):
            P = np.tile(b, (N_GRID, 1)); P[:, j] = g
            S[r] = gp.predict(P)
        curves.append(S)

    lo = min(float(c.min()) for c in curves + [yy.reshape(1, -1)])
    hi = max(float(c.max()) for c in curves + [yy.reshape(1, -1)])
    pad = 0.07 * (hi - lo + 1e-12)

    fig, axes = plt.subplots(1, d, figsize=(3.3 * d, 3.6), squeeze=False)
    for j in range(d):
        S = curves[j]; ax = axes[0, j]
        ax.fill_between(g, np.percentile(S, 10, axis=0), np.percentile(S, 90, axis=0),
                        color=MEAN, alpha=0.18, lw=0,
                        label="effect of the other inputs")
        ax.plot(g, S.mean(0), color=MEAN, label="surrogate")
        ax.scatter(X[:, j], yy, s=22, color=INK2, alpha=0.55, lw=0, zorder=3,
                   label="observations")
        ax.axvline(best[j], color=MARK, lw=1.5, ls="--", zorder=2)
        ax.plot(best[j], yy.max(), marker="o", ms=8, color=MARK, mec="#fcfcfb",
                mew=1.5, zorder=4, label="current best")
        ax.set_xlim(0, 1); ax.set_ylim(lo - pad, hi + pad); ax.set_xlabel(f"x{j}")
        if j == 0:
            ax.set_ylabel(label); ax.legend(fontsize=7.5, loc="best")
        else:
            ax.set_yticklabels([])

    inside = 0
    for j in range(d):
        S = curves[j]
        loq = np.interp(X[:, j], g, np.percentile(S, 10, axis=0))
        hiq = np.interp(X[:, j], g, np.percentile(S, 90, axis=0))
        inside += int(np.sum((yy >= loq) & (yy <= hiq)))
    pct = 100 * inside / (d * len(yy))

    fig.suptitle(f"F{i}  —  the surrogate along each input      "
                 f"({pct:.0f}% of dots fall inside the band, i.e. fully explained "
                 f"by the other inputs)",
                 color=INK, fontsize=11.5, y=1.04, ha="left", x=0.02)
    plt.tight_layout(); plt.show()
    return gp
print("ready")
